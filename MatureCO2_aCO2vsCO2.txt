# Load your data
data <- read.csv("may.csv")

# Extract the group labels (excluding the first column which is gene names)
group_labels <- colnames(data)[-1]

gathercols=c('MayControl', 'MayControl.1','MayControl.2','MayControl.3','MayControl.4','MayControl.5','MayCO2','MayCO2.1','MayCO2.2','MayCO2.3','MayCO2.4')


# Find indices of columns by group
aCO2_cols <-c('MayControl', 'MayControl.1','MayControl.2','MayControl.3','MayControl.4','MayControl.5')
eCO2_cols <-c('MayCO2','MayCO2.1','MayCO2.2','MayCO2.3','MayCO2.4','MayCO2.5')

# Function to compute stats per row
compute_stats <- function(row) {
  aCO2 <- as.numeric(row[aCO2_cols])
  eCO2 <- as.numeric(row[eCO2_cols])
  
  # Mann-Whitney U test (Wilcoxon rank-sum test)
  pval <- wilcox.test(aCO2, eCO2, exact = FALSE)$p.value
  
  
  # Means
  aCO2average <- mean(aCO2, na.rm = TRUE)
  eCO2average <- mean(eCO2, na.rm = TRUE)
  
  # Log2 Fold Change (add pseudocount to avoid log(0))
  log2fc <- log2(abs((eCO2average) / (aCO2average)))
  
  return(c(mean_aCO2 = aCO2average, mean_eCO2 = eCO2average, log2FC = log2fc, p_value = pval))
}

# Apply the function to all rows
results <- t(apply(data, 1, compute_stats))

# Combine with original data
final_data <- cbind(data, results)

# Write to CSV
write.csv(final_data, "may_stats.csv", row.names = FALSE)


#replace the file path with your file path to the document
maydata <- read.table("may_stats.csv", header = TRUE, sep = ",")


#bringing in the required values in
log10.pval.may <- -log10(maydata$p_value)

log2.fc.may <- maydata$log2FC

maydata$color <- with(maydata, ifelse(
  p_value < 0.01 & log2FC > 1, "#FFFF00", # purple
  ifelse(p_value < 0.01 & log2FC < -1.5, "#800080", "grey70") # blue / grey
))

color <- maydata$color

Label <- maydata$Label

#Run the next few lines as a single chunk
plot(log2.fc.may,log10.pval.may, col = color, pch =16, xlab="log2 (fold change)", ylab="-log10 (p-value)", xlim=c(-5,5), ylim=c(0,4), sub = "may control vs elevated CO2", cex.sub = 0.75, font.sub = 2)
abline(h = -log10(0.05), col='red',lwd = 1.5)
abline(v = -log2(2), col='blue',lwd = 1.5)
abline(v = log2(2), col='blue',lwd = 1.5)

sigmaydata <- maydata[which(maydata$p_value < 0.02 & maydata$p_value > 0 & abs(maydata$log2FC) > 1), ]

sigmaydatavals <- subset(sigmaydata, select = -c(p_value, mean_eCO2, mean_aCO2,log2FC, color, Label) )

group_labels <- ifelse(grepl("Control", colnames(sigmaydatavals)), "aCO2", "eCO2")

# Turn into a data.frame for annotation_col
annotation_df <- data.frame(Group = factor(group_labels))
rownames(annotation_df) <- colnames(sigmaydatavals)
annotation_colors_rosa <- list(Group = c("aCO2" = "#1874cd", "eCO2" = "#ffa500"))
my_breaks <- seq(-3, 3, length.out = 101)

pheatmap(sigmaydatavals, 
         main = "Heatmap of Significant Metabolites",
         fontsize_row = 8,
         fontsize_col = 0.000001,
         #annotation_col = sigmaydatavals["Label"], 
         cluster_rows = TRUE,
         cluster_cols = FALSE,
         clustering_distance_rows = "euclidean",
         scale = "row", 
         color = viridis(100, option = "viridis"),
         breaks = my_breaks,
         treeheight_row = 30,
         border_color = NA,
         annotation_col = annotation_df,
         annotation_colors = annotation_colors_rosa
)
